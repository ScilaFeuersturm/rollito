## Concepto

Le pediste a la IA: *"Mostrá la lista de productos de la API y un mensaje si falla"*. Te dio esto. Compila, pero en el emulador se cierra al abrir la pantalla.

Esta vez los problemas son de **hilos, ciclo de vida, eventos y listas**. Son **seis**. Antes de abrir las respuestas, pensá también qué pasa al rotar y al minimizar la app.

## Ejemplo

```kotlin
class ProductosViewModel : ViewModel() {
    val productos = MutableStateFlow(emptyList<Producto>())
    val errores = MutableStateFlow<String?>(null)

    init {
        viewModelScope.launch {
            val json = URL("http://api.tienda.com/productos").readText()
            productos.value = Json.decodeFromString<List<Producto>>(json)
        }
    }
}

class ProductosActivity : ComponentActivity() {
    private val viewModel: ProductosViewModel by viewModels()

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        lifecycleScope.launch {
            viewModel.errores.collect { msg ->
                if (msg != null) Toast.makeText(this@ProductosActivity, msg, Toast.LENGTH_SHORT).show()
            }
        }
        setContent {
            val productos by viewModel.productos.collectAsState()
            LazyColumn {
                items(productos) { producto -> FilaProducto(producto) }
            }
        }
    }
}
```

*Pista:* preguntate en qué hilo corre cada línea y qué pasa con cada colector cuando la Activity se recrea o se va a segundo plano.
